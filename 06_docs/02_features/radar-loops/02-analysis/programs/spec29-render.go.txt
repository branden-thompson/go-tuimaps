// Command spec29 draws specimen 29: real radar under a real warning polygon,
// through go-tuimaps' public calls. Throwaway; kept outside the repository.
//
//	spec29 SPEC29_DIR OUT_PREFIX ORDER_LABEL
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strconv"
	"time"

	tuimaps "github.com/branden-thompson/go-tuimaps"
	"github.com/branden-thompson/go-tuimaps/assets"
)

var captured = time.Date(2026, 9, 23, 14, 28, 0, 0, time.UTC)

type alertDoc struct {
	Features []struct {
		Properties struct {
			Event, Severity, AreaDesc string
		} `json:"properties"`
		Geometry struct {
			Coordinates [][][2]float64 `json:"coordinates"`
		} `json:"geometry"`
	} `json:"features"`
}

func must(err error) {
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

func main() {
	dir, out, order := os.Args[1], os.Args[2], os.Args[3]

	var raw [][]string
	b, err := os.ReadFile(filepath.Join(dir, "n0q-table-raw.json"))
	must(err)
	must(json.Unmarshal(b, &raw))
	table := []tuimaps.TableEntry{}
	for _, r := range raw {
		v, _ := strconv.ParseFloat(r[1], 64)
		R, _ := strconv.Atoi(r[2])
		G, _ := strconv.Atoi(r[3])
		B, _ := strconv.Atoi(r[4])
		table = append(table, tuimaps.TableEntry{Colour: tuimaps.RGB{R: uint8(R), G: uint8(G), B: uint8(B)}, Value: v})
	}

	var ad alertDoc
	b, err = os.ReadFile(filepath.Join(dir, "alert.json"))
	must(err)
	must(json.Unmarshal(b, &ad))
	f := ad.Features[0]
	ring := []tuimaps.LonLat{}
	for _, c := range f.Geometry.Coordinates[0] {
		ring = append(ring, tuimaps.LonLat{Lon: c[0], Lat: c[1]})
	}

	png, err := os.ReadFile(filepath.Join(dir, "radar-crop.png"))
	must(err)

	sizes := []tuimaps.Size{{Cols: 69, Rows: 12}, {Cols: 149, Rows: 38}}
	depths := []struct {
		name string
		d    tuimaps.Depth
		ext  string
	}{{"truecolor", tuimaps.Truecolor, ".ans"}, {"no-colour", tuimaps.NoColour, ".txt"}}

	for _, sz := range sizes {
		for _, dp := range depths {
			m, err := tuimaps.New(tuimaps.WithSize(sz.Cols, sz.Rows), tuimaps.Embed(assets.Tile, assets.MaxZoom))
			must(err)
			must(m.CacheRoot(filepath.Join(dir, "tilecache"), 256<<20))
			must(m.Source("https://tiles.openfreemap.org/planet"))
			m.ColourDepth(dp.d)
			must(m.Ground(tuimaps.RGB{R: 16, G: 22, B: 28}))
			m.PaintGround()
			m.Units(true, true)

			img := tuimaps.Image{
				West: -82.401007, South: 37.949451, East: -81.501007, North: 38.650549,
				Projection: tuimaps.PlateCarree,
				PNG:        png,
				Type:       tuimaps.Type{Preset: "radar", Unit: "dBZ"},
				Table:      table,
			}
			radar := tuimaps.RadarImage("radar", img, captured)
			radar.Credit = "NOAA/NWS NEXRAD via Iowa Environmental Mesonet"
			if os.Getenv("NORADAR") == "" {
				_, err = m.Set(radar)
			}
			must(err)
			_, err = m.Set(tuimaps.Overlay{
				ID: "alerts", Valid: captured, Keeps: 3 * time.Hour, Credit: "National Weather Service",
				Features: []tuimaps.Feature{{Kind: tuimaps.Polygon, Rings: [][]tuimaps.LonLat{ring}, Role: tuimaps.AlertSevere, Label: f.Properties.Event}},
			})
			must(err)
			_, err = m.SetPlaces([]tuimaps.Place{{ID: "hamlin", Name: "Hamlin", At: tuimaps.LonLat{Lon: -82.1029, Lat: 38.2787}, Marker: tuimaps.MarkerRing}})
			must(err)
			must(m.FitTo([]tuimaps.LonLat{{Lon: -82.1029, Lat: 38.2787}}, []string{"alerts"}, 2))
			for i := 0; i < 6; i++ {
				_, err = m.Settle(context.Background())
				must(err)
				if m.Pending() == 0 {
					break
				}
			}
			fr, err := m.Render(sz, captured)
			must(err)
			for _, w := range m.Warnings() {
				fmt.Fprintln(os.Stderr, "warning:", w)
			}
			c, z := m.Centre()
			name := fmt.Sprintf("%s-%dx%d%s", out, sz.Cols, sz.Rows, dp.ext)
			fh, err := os.Create(name)
			must(err)
			fmt.Fprintf(fh, "Specimen 29 %s · %s · %dx%d · %s · %s (%s) %s, %s · radar IEM n0q 2026-09-23 14:28Z · centre %.3f,%.3f z%.2f · status %s\n",
				order, dp.name, sz.Cols, sz.Rows, "Hamlin WV", f.Properties.Event, f.Properties.Severity, f.Properties.AreaDesc, "go-tuimaps public calls", c.Lat, c.Lon, z, fr.Status)
			for _, l := range fr.Lines {
				fmt.Fprintln(fh, l)
			}
			if footer := m.Footer(); footer != "" {
				fmt.Fprintln(fh, footer)
			}
			fh.Close()
			fmt.Println("wrote", name, fr.Status)
			m.Close()
		}
	}
}
